如

<Text style={[{
        fontSize: 16,
        lineHeight: 18,
        numberOfLines: 1,
        flex: 1,

kuikly
直接给组件添加对应的色值属性，themeStyle.c_txt1中用客户端skin_text_main_color色值给Text设置了color属性，kuikly获取客户端特定色值的方式是ctx.buildThemedColor，其中ctx对应的是BasePage的上下文，所以对应的kuikly实现是

Text {
  attr {
    fontSize(32f.pxw())
    lineHeight(36f.pxw())
    flex(1f)
    color(ctx.buildThemedColor("skin_text_main_color"))
  }
}

View {
  backgroundColor(ctx.buildThemedColor("skin_mask_color"))
}

// 引入相关文件
import com.example.business.bizkuikly.data.model.IJCEReqModel
import com.example.business.bizkuikly.data.model.IJCERspModel
import com.tencent.kuikly.core.nvi.serialization.json.JSONObject
// 声明一个定义请求参数的类
internal class GetRecommendReq(
    val songId: Long,
) : IJCEReqModel, IJCERspModel<GetRecommendReq> {
    override fun encode(): JSONObject {
        return JSONObject().apply {
            put("songid", songId)
            put("Source", 1)
            put("extra_info", JSONObject().apply {
                put("needsong_num", "3")
            })
        }
    }

    override fun decode(data: JSONObject): GetRecommendReq {
        return this
    }
}

// 发起请求，引入NetworkInfo模块
import com.example.business.bizkuikly.data.NetworkInfo

val networkInfo = NetworkInfo(
  module = "music.recommend.TrackRelationServer",
  method = "GetSimilarSongs",
  params = GetRecommendReq(curSongId)
    .encode().toString()
)
getPager().networkModule.jsonModuleRequest(networkInfo) { json ->
  // 回调
	val code = json?.optLong("code")
	val data = json?.optJSONObject("data")
	val rsp = data?.let {
		GetDataRsp()
			.decode(it)
	}
	// 对rsp做处理
}

react native
media_list__cover: {
	position: 'relative',
	marginRight: 16,
	width: 60,
	height: 60,
},
kuikly
// BasePage中重写uiDesign方法
override fun uiDesign(): UIDesign {
    return UIDesign(780F, 1624F)
}

View {
	attr {
		positionRelative()
		marginRight(32f.pxw())  // 16 * 2
		width(120f.pxw())       // 60 * 2
		height(120f.pxw())      // 60 * 2
	}
}

18. 版本判断

kuikly
